cmake_minimum_required(VERSION 3.20)
project(FarCryMinimalTest VERSION 1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Platform detection
if(APPLE)
    message(STATUS "Building minimal test for macOS")
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
        message(STATUS "Target: Apple Silicon (ARM64)")
        add_compile_definitions(_CPU_ARM64=1)
    else()
        message(STATUS "Target: Intel Mac (x86_64)")
        add_compile_definitions(_CPU_X86_64=1)
    endif()
    
    add_compile_definitions(__APPLE__=1 __MACH__=1)
    
    # macOS deployment target
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0")
endif()

# Include directories
include_directories(${CMAKE_SOURCE_DIR})

# Test executable
add_executable(farcry_minimal_test test_minimal.cpp)

# Disable memory manager for this test
target_compile_definitions(farcry_minimal_test PRIVATE NOT_USE_CRY_MEMORY_MANAGER=1)

# Compiler warnings
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(farcry_minimal_test PRIVATE
        -Wno-deprecated-declarations
        -Wno-unused-variable
        -Wno-macro-redefined
    )
endif()
