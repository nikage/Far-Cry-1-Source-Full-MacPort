cmake_minimum_required(VERSION 3.20)
project(FarCryMacSilicon VERSION 1.34 LANGUAGES C CXX OBJCXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Platform detection
if(APPLE)
    message(STATUS "Building Far Cry for macOS ARM64 (Apple Silicon)")
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0")
    
    # Apple Silicon optimizations
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
        add_compile_definitions(_CPU_ARM64=1)
        add_compile_options(-mcpu=apple-m1)
        message(STATUS "Enabled Apple Silicon optimizations")
    endif()
    
    # macOS defines
    add_compile_definitions(__APPLE__=1 __MACH__=1 NOT_USE_CRY_MEMORY_MANAGER=1)
    
    # Find frameworks
    find_library(COCOA_FRAMEWORK Cocoa REQUIRED)
    find_library(METAL_FRAMEWORK Metal REQUIRED)
    find_library(CORE_AUDIO_FRAMEWORK CoreAudio REQUIRED)
    find_library(AUDIO_TOOLBOX_FRAMEWORK AudioToolbox REQUIRED)
    find_library(IOKIT_FRAMEWORK IOKit REQUIRED)
    find_library(CORE_FOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
    find_library(CARBON_FRAMEWORK Carbon REQUIRED)
    find_library(APPKIT_FRAMEWORK AppKit REQUIRED)
    
    message(STATUS "All macOS frameworks found successfully")
endif()

# Include directories (no STLPORT for macOS)
include_directories(${CMAKE_SOURCE_DIR})

# Compiler flags
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    add_compile_options(
        -Wno-deprecated-declarations
        -Wno-unused-variable
        -Wno-unused-function
        -Wno-format-security
        -Wno-reorder
        -Wno-switch
        -Wno-macro-redefined
        -Wno-nonportable-include-path
        -Wno-unused-private-field
        -Wno-inconsistent-missing-override
    )
endif()

# Create a simplified Far Cry executable for macOS
set(FARCRY_SOURCES
    test_crysystem_simple.cpp
)

add_executable(FarCryMacSilicon ${FARCRY_SOURCES})

# Link frameworks
if(APPLE)
    target_link_libraries(FarCryMacSilicon PRIVATE
        ${COCOA_FRAMEWORK}
        ${METAL_FRAMEWORK}
        ${CORE_AUDIO_FRAMEWORK}
        ${AUDIO_TOOLBOX_FRAMEWORK}
        ${IOKIT_FRAMEWORK}
        ${CORE_FOUNDATION_FRAMEWORK}
        ${CARBON_FRAMEWORK}
        ${APPKIT_FRAMEWORK}
    )
    
    # Configure as macOS app bundle
    set_target_properties(FarCryMacSilicon PROPERTIES
        MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_BUNDLE_NAME "Far Cry"
        MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
        MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
        MACOSX_BUNDLE_IDENTIFIER "com.crytek.farcry.macsilicon"
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/FARCRY/Info.plist.in"
    )
endif()

# Set output directory
set_target_properties(FarCryMacSilicon PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}
)

# Installation
if(APPLE)
    install(TARGETS FarCryMacSilicon BUNDLE DESTINATION .)
    message(STATUS "Far Cry will be installed as macOS app bundle")
endif()
