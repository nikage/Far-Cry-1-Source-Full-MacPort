cmake_minimum_required(VERSION 3.20)
project(FarCryComplete VERSION 1.34 LANGUAGES C CXX OBJCXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Platform detection
if(APPLE)
    message(STATUS "Building complete Far Cry for macOS ARM64 (Apple Silicon)")
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0")
    
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
        add_compile_definitions(_CPU_ARM64=1)
        add_compile_options(-mcpu=apple-m1)
    endif()
    
    # Find frameworks
    find_library(COCOA_FRAMEWORK Cocoa REQUIRED)
    find_library(METAL_FRAMEWORK Metal REQUIRED)
    find_library(CORE_AUDIO_FRAMEWORK CoreAudio REQUIRED)
    find_library(AUDIO_TOOLBOX_FRAMEWORK AudioToolbox REQUIRED)
    find_library(IOKIT_FRAMEWORK IOKit REQUIRED)
    find_library(CORE_FOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
    find_library(CARBON_FRAMEWORK Carbon REQUIRED)
    find_library(APPKIT_FRAMEWORK AppKit REQUIRED)
endif()

# Include directories
include_directories(${CMAKE_SOURCE_DIR})

# Global macOS settings
if(APPLE)
    add_compile_definitions(
        __APPLE__=1
        __MACH__=1
        NOT_USE_CRY_MEMORY_MANAGER=1
        _PLATFORM_H_=1  # Prevent problematic platform.h
    )
    
    # Use our clean platform header
    add_compile_options(-include ${CMAKE_SOURCE_DIR}/CryCommon/platform_macos.h)
endif()

# Compiler flags
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    add_compile_options(
        -Wno-deprecated-declarations
        -Wno-unused-variable
        -Wno-unused-function
        -Wno-format-security
        -Wno-reorder
        -Wno-switch
        -Wno-macro-redefined
        -Wno-nonportable-include-path
        -Wno-unused-private-field
        -Wno-inconsistent-missing-override
    )
endif()

# Core game source files - build everything into one executable
set(CORE_GAME_SOURCES
    # Main executable
    FARCRY/Main.cpp
    
    # Essential system files
    CrySystem/System.cpp
    CrySystem/Log.cpp
    CrySystem/Timer.cpp
    
    # Input system  
    CryInput/Input.cpp
    CryInput/MacOSInput.cpp
    
    # Core game files
    CryGame/Game.cpp
    CryGame/XPlayer.cpp
    
    # Sound system
    CrySoundSystem/MacOSSound.cpp
    
    # Graphics
    RenderDll/XRenderMetal/MetalRenderer.cpp
)

# Create the complete Far Cry executable
add_executable(FarCryComplete ${CORE_GAME_SOURCES})

# Link frameworks  
if(APPLE)
    target_link_libraries(FarCryComplete PRIVATE
        ${COCOA_FRAMEWORK}
        ${METAL_FRAMEWORK}
        ${CORE_AUDIO_FRAMEWORK}
        ${AUDIO_TOOLBOX_FRAMEWORK}
        ${IOKIT_FRAMEWORK}
        ${CORE_FOUNDATION_FRAMEWORK}
        ${CARBON_FRAMEWORK}
        ${APPKIT_FRAMEWORK}
        "-framework AVFoundation"
        "-framework MetalKit"
        "-framework QuartzCore"
    )
    
    # Configure as macOS app bundle
    set_target_properties(FarCryComplete PROPERTIES
        MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_BUNDLE_NAME "Far Cry Complete"
        MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
        MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
        MACOSX_BUNDLE_IDENTIFIER "com.crytek.farcry.complete"
    )
    
    # Set Objective-C++ for files that need it
    set_source_files_properties(
        CryInput/MacOSInput.cpp
        CrySoundSystem/MacOSSound.cpp
        RenderDll/XRenderMetal/MetalRenderer.cpp
        PROPERTIES COMPILE_FLAGS "-x objective-c++"
    )
endif()

# Install
if(APPLE)
    install(TARGETS FarCryComplete BUNDLE DESTINATION .)
endif()
